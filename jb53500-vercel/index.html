זה מה ש-Claude אמור לכתוב, בערך. שם המודל והפרומפט יהיו שלכם:

const SITE = 'https://username.github.io';
const MODEL = 'gemini-3.8-flash';
const ROLE = 'הפרומפט המרכזי של המוצר';

const cors = {
  'Access-Control-Allow-Origin': SITE,
  'Access-Control-Allow-Methods': 'POST, OPTIONS',
  'Access-Control-Allow-Headers': 'Content-Type',
};

function reply(body, status = 200) {
  return Response.json(body, { status, headers: cors });
}

export default {
  async fetch(request) {
    if (request.method === 'OPTIONS') {
      return new Response(null, { status: 204, headers: cors });
    }
    if (request.method !== 'POST') {
      return reply({ error: 'POST only' }, 405);
    }

    let question = '';
    try {
      const body = await request.json();
      question = String(body.question || '').trim().slice(0, 500);
    } catch (e) {
      return reply({ error: 'bad request' }, 400);
    }
    if (!question) {
      return reply({ error: 'missing question' }, 400);
    }

    const key = process.env.GEMINI_API_KEY;
    if (!key) {
      console.error('GEMINI_API_KEY is missing');
      return reply({ error: 'server is not configured' }, 500);
    }

    const url = 'https://generativelanguage.googleapis.com/v1beta/models/'
      + MODEL + ':generateContent';
    const res = await fetch(url, {
      method: 'POST',
      headers: { 'x-goog-api-key': key, 'Content-Type': 'application/json' },
      body: JSON.stringify({
        system_instruction: { parts: [{ text: ROLE }] },
        contents: [{ parts: [{ text: question }] }],
      }),
    });

    if (!res.ok) {
      console.error('model error', res.status, await res.text());
      return reply({ error: 'model did not answer' }, 502);
    }

    const data = await res.json();
    const parts = data.candidates?.[0]?.content?.parts || [];
    const answer = parts.map((p) => p.text || '').join('').trim();
    return reply({ answer });
  },
};

ובצד של האתר, הכפתור פונה לפונקציה כך:

const API = 'https://my-product.vercel.app/api/ask';

async function ask(question) {
  const res = await fetch(API, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ question }),
  });
  const data = await res.json();
  return data.answer || data.error;
}

CORS עוצר אתרים אחרים, לא את כל העולם. מי שמכיר את כתובת הפונקציה יכול לפנות אליה מתוכנה, בלי דפדפן. לכן מגבילים את אורך הקלט, ועוקבים אחרי השימוש אצל ספק המודל. נחזור לזה במפגש האבטחה.

אם היא נכשלה, הצג לי את השגיאה מתוך הלוג, והסבר אותה בעברית במשפט אחד.

המחבר מקבל את אותה גישה שיש לחשבון שלכם ב-Vercel. הוא יודע גם לפרוס ולשנות הגדרות. נשארים במצב Manual, קוראים כל פעולה, ולא מאשרים "תמיד" לפעולה שמשנה משהו. בקורס הפריסה נשארת דרך GitHub.
כשמשהו לא עובד
+הכתובת /api/hello מחזירה 404
הקובץ לא נמצא בתיקיית api שבשורש הריפו, או שהפריסה עוד לא הסתיימה. בודקים ב-Deployments שהפריסה האחרונה מסומנת Ready, ושהקומיט שלכם מופיע בה. בודקים גם ב-GitHub שהתיקייה api נמצאת ליד index.html, ולא בתוך תיקייה אחרת.
+הפונקציה מחזירה שגיאה 500
פותחים את Logs ב-Vercel, מוצאים את השורה האדומה, ומעתיקים אותה ל-Claude: "זו השגיאה מהלוג של Vercel. הסבר לי מה היא אומרת לפני שאתה מתקן." הסיבה הנפוצה: המשתנה לא הוגדר, או שלא הייתה פריסה חדשה אחרי שהוגדר.
+הכתובת /api/check ממשיכה להחזיר false
או שהשם לא זהה אות באות ל-GEMINI_API_KEY, או שלא הייתה פריסה חדשה אחרי השמירה. בודקים את השם ב-Environment Variables, ועושים Redeploy.
+בקונסול מופיעה שגיאת CORS
הכתובת שכתובה בפונקציה לא זהה לכתובת האתר. בודקים: מתחילה ב-https, נגמרת ב-github.io, בלי שם הריפו ובלי לוכסן בסוף. או שפתחתם את האתר מקובץ מקומי, ואז זו ההתנהגות הנכונה.
+הפריסה נכשלת, והשגיאה מזכירה export או module
חסר בשורש הריפו הקובץ package.json עם השורה "type": "module". חוזרים לשלב 4.
+דחפתי, ואין פריסה חדשה
ב-Vercel, בהגדרות הפרויקט תחת Git, בודקים שהפרויקט מחובר לריפו הנכון. בודקים ב-GitHub שהקומיט באמת הגיע לענף main. אם הכל נראה תקין, פותחים את Deployments ויוצרים פריסה ידנית עם Create Deployment.
+הכתובת מבקשת ממני להתחבר ל-Vercel
בדרך כלל זה אומר שפתחתם כתובת של פריסה מסוימת: ארוכה, עם אותיות ומספרים באמצע. משתמשים בכתובת הקצרה של הפרויקט, שמופיעה בדשבורד תחת Domains.
+האתר מציג הודעת שגיאה במקום תשובה
הפונקציה עובדת, והספק החזיר שגיאה. הסיבה המדויקת כתובה ב-Logs. בשכבה החינמית זו בדרך כלל מכסה לדקה: מחכים רגע ומנסים שוב. אם השגיאה מזכירה שם של מודל, מבקשים מ-Claude לבדוק את שם המודל בתיעוד העדכני.
+עובד מכתובת Vercel, ולא מכתובת GitHub Pages
שתי בדיקות. האחת, שהכפתור פונה לכתובת המלאה של הפונקציה, עם https:// ושם הפרויקט, ולא רק ל-/api/ask. השנייה, ש-GitHub Pages כבר התעדכן: לפעמים הוא מתעדכן דקה או שתיים אחרי Vercel.
+המפתח נדחף בטעות לריפו
מחליפים אותו מיד: ב-Google AI Studio מוחקים את המפתח ויוצרים חדש, מעדכנים את הערך ב-Vercel, ועושים Redeploy. למחוק את השורה מהקוד לא מספיק, כי ההיסטוריה של Git שומרת כל גרסה. מפתח שנחשף פעם אחת נחשב חשוף.
+Claude מבקש את המפתח
לא נותנים. כותבים לו: "המפתח שמור ב-Vercel. קרא אותו בקוד מתוך process.env, ואל תבקש אותו ממני."
